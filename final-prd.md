9.3 Dynamic Pricing Algorithm (TypeScript)
TypeScript
interface CalculateFareInput {
nights: number;
basePricePerNight: number;
baseAdults: number;
selectedAdults: number;
selectedKids: number; // 0 to 12 years are free
extraPersonFee: number; // default ₹500
selectedPaidAddons: { price: number; pricingType: 'flat' | 'per_night' }[];
}
export function calculateFlatelPrice(input: CalculateFareInput){
final prd 10
const { nights, basePricePerNight, baseAdults, selectedAdults, extraPersonFee, selectedPaidAddons } =
input;
// 1. Base Rent with Automatic Duration Discounts
let baseRent = basePricePerNight * nights;
if (nights >= 28) {
baseRent = baseRent * 0.75; // 25% discount for monthly stays
} else if (nights >= 15) {
baseRent = baseRent * 0.85; // 15% discount for 15+ nights
} else if (nights >= 7) {
baseRent = baseRent * 0.90; // 10% discount for weekly stays
}
// 2. Extra Person Fee (Only adults exceeding base capacity; kids 0-12 are free)
const extraAdults = Math.max(0, selectedAdults - baseAdults);
const extraGuestsTotal = extraAdults * extraPersonFee * nights;
// 3. Paid Add-ons Total
const addonsTotal = selectedPaidAddons.reduce((sum, addon) => {
return addon.pricingType === 'per_night'
? sum + (addon.price * nights)
: sum + addon.price;
}, 0);
// 4. Gross Total & 20% Deposit Split
const grossTotal = Math.round(baseRent + extraGuestsTotal + addonsTotal);
const advancePayable = Math.round(grossTotal * 0.20);
const balanceDueOnArrival = grossTotal - advancePayable;
return {
nights,
baseRent,
extraAdults,
extraGuestsTotal,
addonsTotal,
grossTotal,
advancePayable,
balanceDueOnArrival
};
}
